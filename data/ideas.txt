def normalize_audio(audio_data):
    max_val = np.max(np.abs(audio_data))
    if max_val > 0.01:  # Avoid amplifying silence

    return audio_data


6. Use a Speech-Focused Bandpass Filter - Human speech is concentrated in 300Hz-3400Hz. Apply a bandpass filter to remove low-frequency road rumble and high-frequency noise:
from scipy.signal import butter, sosfilt
def speech_bandpass_filter(audio, rate, lowcut=300, highcut=3400, order=5):
    sos = butter(order, [lowcut, highcut], btype='band', fs=rate, output='sos')
    return sosfilt(sos, audio)
